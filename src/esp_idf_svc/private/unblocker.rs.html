15pub struct Unblocker<T>
16where
17    T: Send + 'static,
18{
19    receiver: Receiver<T>,

22impl<T> Unblocker<T>
23where
24    T: Send + 'static,
25{
26    /// Create a new Unblocker
27    ///
28    /// # Arguments
29    ///
30    /// * `task_name` - The name of the task.
31    /// * `stack_size` - The size of the stack for the task, in bytes.
32    /// * `priority` - The priority of the task.
33    /// * `pin_to_core` - The core to pin the task to, if any.
34    /// * `worker` - The function to run in the task. A channel will be passed to the worker with all events.
35    ///   The channel will be closed when the Unblocker is dropped, at which point the worker should exit.
36    pub fn new<F>(
37        task_name: &'static CStr,
38        stack_size: usize,
39        priority: Option<u8>,
40        pin_to_core: Option<Core>,
41        mut worker: F,
42    ) -> Result<Self, EspError>
43    where
44        F: FnOnce(Arc<Channel<T>>) + Send + 'static,
45        T: Send + 'static,
46    {
47        let (channel, receiver) = Channel::new();
48
49        let worker: Box<Box<dyn FnOnce() + Send + 'static>> = Box::new(Box::new(move || {
50            worker(channel);
51        }));
52
53        let worker: *mut Box<dyn FnOnce() + Send + 'static> = Box::into_raw(worker);
54
55        let _task = unsafe {
56            task::create(
57                Self::work,
58                task_name,
59                stack_size,
60                worker as *mut _,
61                priority.unwrap_or(6),
62                pin_to_core,
63            )
64        }
65        .inspect_err(|_| unsafe {
66            // Avoid memory leak if task creation fails
67            Box::from_raw(worker);
68        })?;
69
70        Ok(Self { receiver })
71    }
72
73    pub async fn exec_in_out(&mut self) -> Option<&mut T> {
74        self.receiver.get_shared_async().await
75    }
76
77    pub async fn do_exec(&mut self) {
78        self.receiver.done();
79    }
80
81    extern "C" fn work(arg: *mut core::ffi::c_void) {
82        {
83            let worker: Box<Box<dyn FnOnce() + Send + 'static>> =
84                unsafe { Box::from_raw(arg as *mut _) };
85
86            worker();
87        }
88        unsafe {
89            // FreeRTOS tasks must delete themselves. Returning from the task function
90            // without deleting the task will cause a crash.
91            // This function will immediately stop running. This means we have to ensure
92            // that all memory is freed before calling vTaskDelete.
93            // Fortunately, this function does not manage any memory at this point.
94            // The scope above guarantees that all relevant variables are dropped at this point.
95            // See https://www.freertos.org/implementing-a-FreeRTOS-task.html
96            task::destroy(core::ptr::null_mut());
97        }
98    }
99}

101impl<T> Drop for Unblocker<T>
102where
103    T: Send + 'static,
104{
105    fn drop(&mut self) {
106        // This should cause the worker task to exit
107        self.receiver.done();
108    }
