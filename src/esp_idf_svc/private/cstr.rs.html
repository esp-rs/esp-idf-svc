12pub fn set_str(buf: &mut [u8], s: &str) -> Result<(), EspError> {
13    if s.len() >= buf.len() {
14        return Err(EspError::from_infallible::<ESP_ERR_INVALID_SIZE>());
15    }
16
17    buf[..s.len()].copy_from_slice(s.as_bytes());
18    buf[s.len()] = 0;
19
20    Ok(())
21}

23pub fn set_str_no_termination_requirement(buf: &mut [u8], s: &str) -> Result<(), EspError> {
24    if s.len() > buf.len() {
25        return Err(EspError::from_infallible::<ESP_ERR_INVALID_SIZE>());
26    }
27
28    buf[..s.len()].copy_from_slice(s.as_bytes());
29    if buf.len() != s.len() {
30        buf[s.len()] = 0;
31    }
32
33    Ok(())
34}

46pub fn from_cstr_fallible(buf: &[u8]) -> Result<&str, Utf8Error> {
47    // We have to find the first '\0' ourselves, because the passed buffer might
48    // be wider than the ASCIIZ string it contains
49    let len = buf.iter().position(|e| *e == 0).unwrap() + 1;
50
51    unsafe { CStr::from_bytes_with_nul_unchecked(&buf[0..len]) }.to_str()
52}
53
54pub fn from_cstr(buf: &[u8]) -> &str {
55    from_cstr_fallible(buf).unwrap()
56}

58/// Convert buffer of characters to heapless string, allowing either
59/// the full buffer (without terminating 0) or just a part of it (terminated by 0)
60pub fn array_to_heapless_string_failible<const N: usize>(
61    arr: [u8; N],
62) -> Result<heapless::String<N>, Utf8Error> {
63    let len = arr.iter().position(|e| *e == 0).unwrap_or(N);
64    heapless::String::from_utf8(heapless::Vec::from_slice(&arr[0..len]).unwrap())
65}
66
67pub fn array_to_heapless_string<const N: usize>(arr: [u8; N]) -> heapless::String<N> {
68    array_to_heapless_string_failible(arr).unwrap()
69}

120/// str to cstr, will be truncated if str is larger than buf.len() - 1
121///
122/// # Panics
123///
124/// * Panics if buffer is empty.
125pub fn cstr_from_str_truncating<'a>(rust_str: &str, buf: &'a mut [u8]) -> &'a CStr {
126    assert!(!buf.is_empty());
127
128    let max_str_size = buf.len() - 1; // account for NUL
129    let truncated_str = &rust_str[..max_str_size.min(rust_str.len())];
130    buf[..truncated_str.len()].copy_from_slice(truncated_str.as_bytes());
131    buf[truncated_str.len()] = b'\0';
132
133    CStr::from_bytes_with_nul(&buf[..truncated_str.len() + 1]).unwrap()
134}

136/// Convert slice of rust strs to NULL-terminated fixed size array of c string pointers
137///
138/// # Panics
139///
140/// * Panics if cbuf is empty.
141/// * Panics if N is <= 1
142pub fn cstr_arr_from_str_slice<const N: usize>(
143    rust_strs: &[&str],
144    mut cbuf: &mut [u8],
145) -> Result<[*const c_char; N], EspError> {
146    assert!(N > 1);
147    assert!(!cbuf.is_empty());
148
149    // ensure last element stays NULL
150    if rust_strs.len() > N - 1 {
151        return Err(EspError::from_infallible::<ESP_ERR_INVALID_SIZE>());
152    }
153
154    let mut cstrs = [core::ptr::null(); N];
155
156    for (i, s) in rust_strs.iter().enumerate() {
157        let max_str_size = cbuf.len() - 1; // account for NUL
158        if s.len() > max_str_size {
159            return Err(EspError::from_infallible::<ESP_ERR_INVALID_SIZE>());
160        }
161        cbuf[..s.len()].copy_from_slice(s.as_bytes());
162        cbuf[s.len()] = b'\0';
163        let cstr = CStr::from_bytes_with_nul(&cbuf[..s.len() + 1]).unwrap();
164        cstrs[i] = cstr.as_ptr();
165
166        cbuf = &mut cbuf[s.len() + 1..];
167    }
168
169    Ok(cstrs)
170}
